import math as m import random as r import tkinter as tk import tkinter.font as tkf W, H = 800, 800 DURATION = 10 TICK = 16 ANIM = 38 LIFT = 32 GAP = 3 BG = "#000000" FONT_FALLBACKS = ("Georgia", "Times New Roman", "Arial") SEED = 17 random.seed(SEED) PHRASES = [ "I LOVE YOU", "I love you", "Te amo", "Ti amo", "AMOUR", "Je t'aime", "Ich liebe dich", "愛してる", "사랑해", "我爱你", "أحبك", "Я тебя люблю", "Seni seviyorum", "Nakupenda", "Mahal kita", "Eu te amo", "Я люблю тебя", "Ik hou van jou", "Kocham cię", "Amo", "Te quiero", "Ljubim te", "Volim te", "Minä rakastan sinua", "Jeg elsker deg", "Jeg elsker dig", "Szeretlek", "Kimi o aishiteru", "Aš tave myliu", "Ti tengu caru", "Aloha wau ia oe", "Wo ai ni", "Ya lyublyu tebya", "Ana uhibbuka", "Main tumse pyar karta hoon", "Miluji tě", "Ik hou van je", "T'estimo", "Ek het jou lief", "Maite zaitut", "Te iubesc", "S'agapo", "Seni seviyorum", "Ljubim te", "Amo-te" ] GREEN_WORDS = [ "LOVE", "AMOR", "TE AMO", "I LOVE YOU", "FOREVER", "MY LOVE", "ALWAYS", "AMOUR", "TI AMO", "BELOVED" ] root = tk.Tk() root.title("1 Rose, Every Language") root.geometry(f"{W}x{H}") root.configure(bg=BG) root.resizable(False, False) canvas = tk.Canvas( root, width=W, height=H, bg=BG, highlightthickness=0 ) canvas.pack() items = [] RED = [ "#65001A", "#800020", "#99002D", "#B0003A", "#C40045", "#D51A54", "#E52B65", "#F04A7B" ] GREEN = [ "#164B25", "#20632D", "#347D3C", "#559B55", "#79B76B" ] def add_text(x, y, text, color, size=8): font = ("Georgia", size, "bold") item = canvas.create_text( x, y, text=text, fill=BG, font=font, anchor="center" ) items.append((item, color)) def draw_title(): title = "1 Rose, Every Language" cx = W / 2 cy = 83 radius = 300 for i, letter in enumerate(title): angle = math.pi + ( math.pi * i / (len(title) - 1) ) x = cx + math.cos(angle) * radius y = cy + math.sin(angle) * 48 add_text( x, y, letter, "#D20B38", 25 ) def rose_width(y): R = 190 if abs(y) > R: return 0 base = math.sqrt( max(0, 1 - (y / R) ** 2) ) wave = ( 0.88 + 0.12 * math.cos( 8 * math.acos( max(-1, min(1, y / R)) ) ) ) taper = 1.0 if y > 85: taper = max( 0.08, 1 - (y - 85) / 120 ) return R * base * wave * taper def create_rose(): cx = W / 2 cy = 310 y = -175 while y <= 175: half_width = rose_width(y) if half_width > 12: x = -half_width + 4 while x < half_width - 5: phrase = random.choice(PHRASES) size = random.choice([6, 7, 8, 9]) spacing = random.randint(34, 52) add_text( cx + x, cy + y, phrase, random.choice(RED), size ) x += spacing y += 11 def create_stem(): cx = W / 2 for i in range(24): y = 490 + i * 10 x = cx + m.sin(i * 0.25) * 4 add_text( x, y, random.choice(GREEN_WORDS), random.choice(GREEN), 7 ) def create_leaves(): cx = W / 2 leaves = [ (cx - 100, 535, -1), (cx + 100, 550, 1), (cx - 80, 590, -1), (cx + 80, 600, 1) ] for lx, ly, side in leaves: for i in range(9): t = i / 8 x = lx + side * t * 65 y = ly - t * 35 add_text( x, y, random.choice(GREEN_WORDS), random.choice(GREEN), 7 ) def create_roots(): cx = W / 2 base_y = 730 branches = [ (-160, 785), (-120, 770), (-75, 790), (-35, 780), (0, 800), (35, 780), (75, 790), (120, 770), (160, 785) ] for end_x, end_y in branches: for i in range(6): t = i / 6 x = cx + end_x * t y = base_y + (end_y - base_y) * t add_text( x, y, random.choice(GREEN_WORDS), random.choice(GREEN), 7 ) def animate(): total = len(items) for i, (item, color) in enumerate(items): def reveal(target=item, shade=color): canvas.itemconfigure( target, fill=shade ) root.after(i * ANIM, reveal) root.after( total * ANIM + 2500, restart ) def restart(): for item, color in items: canvas.itemconfigure( item, fill=BG ) root.after(500, animate) draw_title() create_rose() create_leaves() create_stem() create_roots() root.after(500, animate) root.mainloop()
